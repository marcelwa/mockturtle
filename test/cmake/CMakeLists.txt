cmake_minimum_required(VERSION 3.25)
project(mockturtle_cmake_contract LANGUAGES C CXX)

enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_compile_definitions(MOCKTURTLE_PARENT_SENTINEL)
get_directory_property(parent_options COMPILE_OPTIONS)
get_directory_property(parent_definitions COMPILE_DEFINITIONS)
get_directory_property(parent_includes INCLUDE_DIRECTORIES)

# Empty targets must not substitute for mockturtle's vendored dependencies.
foreach(name parallel_hashmap fmt kitty rang range lorina json percy bill libabcsat libabcesop nauty matplot)
  add_library(${name} INTERFACE)
endforeach()

if(MOCKTURTLE_SOURCE)
  add_subdirectory("${MOCKTURTLE_SOURCE}" mockturtle SYSTEM)
  if(MOCKTURTLE_BUILD_EXAMPLES OR MOCKTURTLE_BUILD_TESTS OR MOCKTURTLE_BUILD_EXPERIMENTS OR MOCKTURTLE_INSTALL)
    message(FATAL_ERROR "Embedded mockturtle enabled standalone work")
  endif()
  foreach(archive mockturtle_abcsat mockturtle_abcesop)
    get_target_property(kind ${archive} TYPE)
    get_target_property(pic ${archive} POSITION_INDEPENDENT_CODE)
    if(NOT kind STREQUAL "STATIC_LIBRARY" OR NOT pic)
      message(FATAL_ERROR "${archive} must be a PIC static library")
    endif()
  endforeach()
else()
  find_package(mockturtle CONFIG REQUIRED)
endif()

get_directory_property(after_options COMPILE_OPTIONS)
get_directory_property(after_definitions COMPILE_DEFINITIONS)
get_directory_property(after_includes INCLUDE_DIRECTORIES)
if(NOT CMAKE_CXX_STANDARD EQUAL 20 OR CMAKE_EXPORT_COMPILE_COMMANDS OR NOT BUILD_SHARED_LIBS
   OR NOT parent_options STREQUAL after_options
   OR NOT parent_definitions STREQUAL after_definitions
   OR NOT parent_includes STREQUAL after_includes)
  message(FATAL_ERROR "mockturtle changed parent configuration")
endif()
foreach(name mockturtle sat esop all)
  if(NOT TARGET mockturtle::${name})
    message(FATAL_ERROR "Missing public target mockturtle::${name}")
  endif()
endforeach()

add_executable(base base.cc)
target_link_libraries(base PRIVATE mockturtle::mockturtle)
add_test(NAME base COMMAND base)
if(NOT BASE_ONLY)
  foreach(component sat esop)
    add_executable(${component} main.cc ${component}.cc)
    target_compile_definitions(${component} PRIVATE CHECK_${component})
    target_link_libraries(${component} PRIVATE mockturtle::${component})
    add_test(NAME ${component} COMMAND ${component})
  endforeach()
  add_library(combined SHARED sat.cc esop.cc)
  set_target_properties(combined PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
  target_link_libraries(combined PRIVATE mockturtle::sat mockturtle::esop)
  add_executable(shared main.cc)
  target_compile_definitions(shared PRIVATE CHECK_sat CHECK_esop)
  target_link_libraries(shared PRIVATE combined)
  add_test(NAME shared COMMAND shared)
  add_executable(aggregate main.cc sat.cc esop.cc)
  target_compile_definitions(aggregate PRIVATE CHECK_sat CHECK_esop)
  target_link_libraries(aggregate PRIVATE mockturtle::all)
  add_test(NAME aggregate COMMAND aggregate)
endif()
