# Public templates include these headers even without invoking a compiled backend.
add_library(mockturtle_dependencies INTERFACE)
set_property(TARGET mockturtle_dependencies PROPERTY EXPORT_NAME _dependencies)
set(mockturtle_header_directories parallel-hashmap fmt kitty rang lorina json percy bill abcsat abcesop)
if(MOCKTURTLE_ENABLE_MATPLOTLIB)
  list(APPEND mockturtle_header_directories matplot)
endif()
foreach(directory IN LISTS mockturtle_header_directories)
  target_include_directories(mockturtle_dependencies SYSTEM INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/${directory}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/mockturtle/dependencies/${directory}>)
endforeach()
target_compile_definitions(mockturtle_dependencies INTERFACE FMT_HEADER_ONLY ABC_NAMESPACE=pabc)
# fmt requires the compiler to interpret source and execution text as UTF-8.
target_compile_options(mockturtle_dependencies INTERFACE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")

# ABC uses LLP64 on Windows and LP64 on Unix; host CPU names are insufficient when cross compiling.
add_library(mockturtle_abc_platform INTERFACE)
set_property(TARGET mockturtle_abc_platform PROPERTY EXPORT_NAME _abc_platform)
if(WIN32)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    target_compile_definitions(mockturtle_abc_platform INTERFACE NT64)
  else()
    target_compile_definitions(mockturtle_abc_platform INTERFACE NT)
  endif()
  target_compile_definitions(mockturtle_abc_platform INTERFACE NOMINMAX WIN32_NO_DLL _CRT_SECURE_NO_WARNINGS)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  target_compile_definitions(mockturtle_abc_platform INTERFACE LIN64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_compile_definitions(mockturtle_abc_platform INTERFACE LIN)
else()
  message(FATAL_ERROR "Unsupported ABC pointer size: ${CMAKE_SIZEOF_VOID_P}")
endif()
target_compile_definitions(mockturtle_abc_platform INTERFACE ABC_NO_USE_READLINE)
target_link_libraries(mockturtle_dependencies INTERFACE mockturtle_abc_platform)

find_package(Threads REQUIRED)
add_subdirectory(abcsat)
add_subdirectory(abcesop)
if(MOCKTURTLE_ENABLE_NAUTY)
  if(WIN32 OR CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Bundled Nauty requires a native Unix configure environment")
  endif()
  enable_language(C)
  add_subdirectory(nauty)
  target_include_directories(mockturtle_dependencies SYSTEM INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/nauty/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/mockturtle/dependencies/nauty>)
  target_compile_definitions(mockturtle_dependencies INTERFACE MOCKTURTLE_ENABLE_NAUTY)
else()
  target_compile_definitions(mockturtle_dependencies INTERFACE DISABLE_NAUTY)
endif()
mockturtle_find_optional_dependencies()
